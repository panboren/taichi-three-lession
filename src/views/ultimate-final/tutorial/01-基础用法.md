# 教程1: 基础用法 - 最简单的集成方式

## 概述

本教程介绍如何使用最基础的配置将 Taichi.js 与 Three.js 集成。这是最简单的入门案例,适合初学者快速上手。

## 目标

- 创建一个简单的 Taichi 模拟引擎
- 使用桥接器将计算结果渲染到 Three.js 场景
- 理解基础的工作流程

## 代码实现

```typescript
import * as THREE from 'three'
import { UltimateTaichiThreeBridge } from '../js/UltimateTaichiThreeBridge'

// 1. 创建 Three.js 基础场景
const scene = new THREE.Scene()
const camera = new THREE.PerspectiveCamera(75, window.innerWidth / window.innerHeight, 0.1, 1000)
const renderer = new THREE.WebGLRenderer()
renderer.setSize(window.innerWidth, window.innerHeight)
document.body.appendChild(renderer.domElement)

camera.position.z = 5

// 2. 创建一个简单的 Taichi 引擎 (模拟)
function createSimpleTaichiEngine() {
  return {
    // 这个方法返回像素数据
    async getPixelData() {
      const width = 512
      const height = 512
      const data = new Uint8Array(width * height * 4)

      // 创建一个简单的动态图案
      const time = performance.now() * 0.001
      for (let y = 0; y < height; y++) {
        for (let x = 0; x < width; x++) {
          const i = (y * width + x) * 4
          const nx = (x / width) - 0.5
          const ny = (y / height) - 0.5
          const dist = Math.sqrt(nx * nx + ny * ny)

          // 创建圆形波纹效果
          const wave = Math.sin(dist * 20 - time * 3)

          data[i] = Math.floor(127 + 127 * wave)     // R
          data[i + 1] = Math.floor(100 + 100 * Math.sin(time))  // G
          data[i + 2] = Math.floor(200 - 100 * wave) // B
          data[i + 3] = 255                          // A
        }
      }

      return data.buffer
    },

    // 更新方法
    update(time?: number) {
      // 可以在这里添加更新逻辑
    }
  }
}

// 3. 创建桥接器
async function initBridge() {
  const taichiEngine = createSimpleTaichiEngine()

  const bridge = new UltimateTaichiThreeBridge({
    renderer,
    scene,
    camera,
    width: 512,
    height: 512,
    mode: 'canvas-texture', // 使用最简单的 CanvasTexture 模式
    autoDetectMode: false,   // 关闭自动检测
    performance: {
      enabled: true,         // 启用性能监控
      updateInterval: 500    // 每500ms更新一次
    },
    onPerformanceUpdate: (metrics) => {
      console.log('FPS:', metrics.fps.toFixed(1))
    }
  })

  // 初始化桥接器
  await bridge.init(taichiEngine)

  return bridge
}

// 4. 运行
async function main() {
  const bridge = await initBridge()

  // 渲染循环
  function animate(time: number) {
    requestAnimationFrame(animate)

    // 更新桥接器 (这会触发 Taichi 计算和数据传输)
    bridge.step(time)

    // Three.js 渲染
    renderer.render(scene, camera)
  }

  animate(0)
}

// 启动
main()

// 清理
window.addEventListener('beforeunload', () => {
  bridge.dispose()
  renderer.dispose()
})
```

## 关键点说明

### 1. Taichi 引擎接口

```typescript
interface TaichiEngine {
  getPixelData(): Promise<ArrayBuffer>  // 必需:返回像素数据
  update?(time?: number): void          // 可选:更新逻辑
}
```

### 2. 桥接器配置

```typescript
{
  renderer,           // Three.js 渲染器 (必需)
  scene,              // Three.js 场景 (必需)
  camera,             // Three.js 相机 (必需)
  width: 512,         // 纹理宽度 (必需)
  height: 512,        // 纹理高度 (必需)
  mode: 'canvas-texture', // 传输模式
  autoDetectMode: false,  // 是否自动检测最优模式
  performance: {      // 性能监控配置
    enabled: true,
    updateInterval: 500
  }
}
```

### 3. 渲染循环

```typescript
function animate(time: number) {
  requestAnimationFrame(animate)

  // 1. 更新桥接器 (计算 + 传输)
  bridge.step(time)

  // 2. 渲染 Three.js 场景
  renderer.render(scene, camera)
}
```
