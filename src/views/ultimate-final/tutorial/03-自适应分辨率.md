# 教程3: 自适应分辨率 - 动态性能优化

## 概述

自适应分辨率可以根据实时性能自动调整渲染分辨率,在保证流畅度的同时最大化视觉效果。本教程讲解如何配置和使用自适应分辨率功能。

## 工作原理

```
性能监控 ──► FPS 检测 ──► 分辨率调整 ──► 重新渲染
    │            │            │
    │            ▼            │
    └────── 阈值判断 ◄─────────┘
             (升/降)
```

## 分辨率调整逻辑

```typescript
// 桥接器内部逻辑
private async adjustAdaptiveResolution(): Promise<void> {
  const metrics = this.performanceMonitor.getMetrics()
  const currentFps = metrics.avgFps
  const targetFps = this.options.targetFps

  let newWidth = this.currentResolution.width
  let newHeight = this.currentResolution.height

  // 性能不足,降低分辨率
  if (currentFps < targetFps * 0.8) {
    newWidth = Math.max(
      this.options.minResolution,
      Math.floor(newWidth * this.options.resolutionScaleFactor)
    )
    newHeight = Math.max(
      this.options.minResolution,
      Math.floor(newHeight * this.options.resolutionScaleFactor)
    )
  }
  // 性能充足,提高分辨率
  else if (currentFps > targetFps * 1.2) {
    newWidth = Math.min(
      this.options.maxResolution,
      Math.floor(newWidth / this.options.resolutionScaleFactor)
    )
    newHeight = Math.min(
      this.options.maxResolution,
      Math.floor(newHeight / this.options.resolutionScaleFactor)
    )
  }

  // 应用新分辨率
  if (newWidth !== this.currentResolution.width ||
      newHeight !== this.currentResolution.height) {
    await this.setSize(newWidth, newHeight)
  }
}
```

## 实例1: 基础自适应

```typescript
import * as THREE from 'three'
import { UltimateTaichiThreeBridge } from '../js/UltimateTaichiThreeBridge'

// 创建场景
const scene = new THREE.Scene()
const camera = new THREE.PerspectiveCamera(75, window.innerWidth / window.innerHeight, 0.1, 1000)
const renderer = new THREE.WebGLRenderer()
renderer.setSize(window.innerWidth, window.innerHeight)
document.body.appendChild(renderer.domElement)

camera.position.z = 5

// 创建动态 Taichi 引擎 (可变计算负载)
function createDynamicEngine() {
  let complexity = 0.5  // 计算复杂度 0-1

  return {
    async getPixelData() {
      const width = 512
      const height = 512
      const data = new Uint8Array(width * height * 4)
      const time = performance.now() * 0.001

      // 根据复杂度动态调整计算量
      const iterations = Math.floor(10 + complexity * 100)

      for (let i = 0; i < width * height; i++) {
        const i4 = i * 4
        const x = i % width
        const y = Math.floor(i / width)
        const nx = x / width - 0.5
        const ny = y / height - 0.5

        let value = 0
        for (let j = 0; j < iterations; j++) {
          const angle = j * 0.1 + time
          value += Math.sin(nx * angle + ny * angle)
        }
        value /= iterations

        data[i4] = Math.floor(127 + 127 * value)
        data[i4 + 1] = Math.floor(127 + 127 * Math.sin(time))
        data[i4 + 2] = Math.floor(127 + 127 * Math.cos(time))
        data[i4 + 3] = 255
      }

      return data.buffer
    },

    setComplexity(val: number) {
      complexity = Math.max(0, Math.min(1, val))
    }
  }
}

const taichiEngine = createDynamicEngine()

// 创建带自适应的桥接器
const bridge = new UltimateTaichiThreeBridge({
  renderer,
  scene,
  camera,
  width: 512,
  height: 512,
  maxResolution: 1024,
  minResolution: 256,
  enableAdaptiveResolution: true,
  targetFps: 60,
  onPerformanceUpdate: (metrics) => {
    console.log(
      `FPS: ${metrics.fps.toFixed(1)} | ` +
      `分辨率: ${metrics.currentResolution.width}x${metrics.currentResolution.height} | ` +
      `趋势: ${metrics.predictedTrend}`
    )
  }
})

await bridge.init(taichiEngine)

// 渲染循环
function animate(time: number) {
  requestAnimationFrame(animate)
  bridge.step(time)
  renderer.render(scene, camera)
}

animate(0)

// 测试: 动态增加计算复杂度
let testTime = 0
setInterval(() => {
  testTime++
  const complexity = (Math.sin(testTime * 0.1) + 1) / 2  // 0-1 循环
  taichiEngine.setComplexity(complexity)
  console.log(`计算复杂度: ${(complexity * 100).toFixed(0)}%`)
}, 1000)
```

## 实例3: 性能预测与预调整

```typescript
// 创建增强的自适应控制器
class AdaptiveResolutionController {
  private bridge: UltimateTaichiThreeBridge
  private predictionHistory: Array<{ time: number, fps: number }> = []

  constructor(bridge: UltimateTaichiThreeBridge) {
    this.bridge = bridge
  }

  // 预测未来性能
  predictFuturePerformance(steps: number = 10): number {
    const history = this.predictionHistory.slice(-20)
    if (history.length < 5) return 60

    // 简单的线性预测
    const n = history.length
    const x = Array.from({ length: n }, (_, i) => i)
    const y = history.map(h => h.fps)

    const sumX = x.reduce((a, b) => a + b, 0)
    const sumY = y.reduce((a, b) => a + b, 0)
    const sumXY = x.reduce((sum, xi, i) => sum + xi * y[i], 0)
    const sumXX = x.reduce((sum, xi) => sum + xi * xi, 0)

    const slope = (n * sumXY - sumX * sumY) / (n * sumXX - sumX * sumX)
    const intercept = (sumY - slope * sumX) / n

    const predicted = slope * (n + steps) + intercept
    return Math.max(0, Math.min(120, predicted))
  }

  // 预调整分辨率
  async preAdjustResolution(): Promise<void> {
    const predictedFps = this.predictFuturePerformance()
    const targetFps = this.bridge.options.targetFps
    const metrics = this.bridge.getMetrics()

    // 如果预测性能将下降,提前降低分辨率
    if (predictedFps < targetFps * 0.9 && metrics.avgFps > targetFps * 1.1) {
      const newWidth = Math.floor(metrics.currentResolution.width * 0.9)
      const newHeight = Math.floor(metrics.currentResolution.height * 0.9)

      console.log(`预调整: 预测 FPS 将降至 ${predictedFps.toFixed(1)}`)
      await this.bridge.setSize(newWidth, newHeight)
    }
  }

  // 更新历史
  update(fps: number): void {
    this.predictionHistory.push({
      time: performance.now(),
      fps
    })

    // 保持最近 100 个数据点
    if (this.predictionHistory.length > 100) {
      this.predictionHistory.shift()
    }
  }
}

// 使用预调整控制器
const adaptiveController = new AdaptiveResolutionController(bridge)

// 在性能更新时更新预测
bridge.options.onPerformanceUpdate = (metrics) => {
  adaptiveController.update(metrics.fps)

  const predicted = adaptiveController.predictFuturePerformance(10)
  console.log(`当前 FPS: ${metrics.fps.toFixed(1)} | 预测 FPS: ${predicted.toFixed(1)}`)

  // 执行预调整
  adaptiveController.preAdjustResolution()
}
```

## 实例4: 多级自适应策略

```typescript
class MultiLevelAdaptiveStrategy {
  private bridge: UltimateTaichiThreeBridge
  private levels = [
    { width: 2048, height: 2048, quality: 'ultra', priority: 1 },
    { width: 1536, height: 1536, quality: 'high', priority: 2 },
    { width: 1024, height: 1024, quality: 'medium', priority: 3 },
    { width: 768, height: 768, quality: 'low', priority: 4 },
    { width: 512, height: 512, quality: 'minimal', priority: 5 }
  ]

  constructor(bridge: UltimateTaichiThreeBridge) {
    this.bridge = bridge
  }

  // 根据性能选择最佳级别
  selectLevel(fps: number, targetFps: number): typeof this.levels[0] {
    const ratio = fps / targetFps

    if (ratio >= 1.5) return this.levels[0]  // ultra
    if (ratio >= 1.2) return this.levels[1]  // high
    if (ratio >= 0.9) return this.levels[2]  // medium
    if (ratio >= 0.7) return this.levels[3]  // low
    return this.levels[4]                    // minimal
  }

  // 应用策略
  async apply(fps: number): Promise<void> {
    const targetFps = this.bridge.options.targetFps
    const level = this.selectLevel(fps, targetFps)
    const metrics = this.bridge.getMetrics()

    // 只在必要时调整
    if (metrics.currentResolution.width !== level.width) {
      console.log(`切换到 ${level.quality} 质量 (${level.width}x${level.height})`)
      await this.bridge.setSize(level.width, level.height)
    }
  }
}

const strategy = new MultiLevelAdaptiveStrategy(bridge)

bridge.options.onPerformanceUpdate = (metrics) => {
  strategy.apply(metrics.avgFps)
}
```

## 监控和可视化

```typescript
// 创建性能图表
function createPerformanceChart() {
  const canvas = document.createElement('canvas')
  canvas.width = 400
  canvas.height = 200
  canvas.style.position = 'fixed'
  canvas.style.bottom = '10px'
  canvas.style.right = '10px'
  canvas.style.background = 'rgba(0,0,0,0.8)'
  canvas.style.border = '1px solid #333'
  document.body.appendChild(canvas)

  const ctx = canvas.getContext('2d')!
  const history: Array<{ fps: number, resolution: number }> = []

  return {
    update(metrics: UltimatePerformanceMetrics) {
      history.push({
        fps: metrics.fps,
        resolution: metrics.currentResolution.width
      })

      // 保持最近 300 帧数据
      if (history.length > 300) history.shift()

      // 绘制
      ctx.fillStyle = 'rgba(0,0,0,0.8)'
      ctx.fillRect(0, 0, canvas.width, canvas.height)

      // 绘制目标 FPS 线
      const targetY = canvas.height - (60 / 120) * canvas.height
      ctx.strokeStyle = '#4CAF50'
      ctx.beginPath()
      ctx.moveTo(0, targetY)
      ctx.lineTo(canvas.width, targetY)
      ctx.stroke()

      // 绘制 FPS 曲线
      ctx.strokeStyle = '#00ff88'
      ctx.beginPath()
      history.forEach((h, i) => {
        const x = (i / 300) * canvas.width
        const y = canvas.height - (h.fps / 120) * canvas.height
        if (i === 0) ctx.moveTo(x, y)
        else ctx.lineTo(x, y)
      })
      ctx.stroke()

      // 绘制分辨率标记
      ctx.fillStyle = '#fff'
      ctx.font = '12px monospace'
      ctx.fillText(
        `FPS: ${metrics.fps.toFixed(1)} | ` +
        `分辨率: ${metrics.currentResolution.width}x${metrics.currentResolution.height}`,
        10, 20
      )
    }
  }
}

const chart = createPerformanceChart()

bridge.options.onPerformanceUpdate = (metrics) => {
  chart.update(metrics)
}
```

## 最佳实践

1. **合理的分辨率范围**:
   - 最低分辨率不应低于 256x256
   - 最高分辨率取决于设备性能
   - 建议设置: min=256, max=2048

2. **调整因子**:
   - `resolutionScaleFactor: 0.9` (每次调整 10%)
   - 太激进会导致闪烁,太保守反应慢

3. **目标 FPS**:
   - 60 FPS 是标准目标
   - 移动设备可考虑 30-45 FPS
   - VR 需要 90+ FPS

4. **结合传输模式**:
   - 性能不足时先降级传输模式
   - 再调整分辨率
   - 两者的组合效果最好
